cmake_minimum_required(VERSION 3.18.1)
project(clab C)
set(CMAKE_C_STANDARD 11)
file(GLOB PICOC_SRC picoc/*.c picoc/cstdlib/*.c picoc/platform/*.c)
add_library(clab SHARED clab_jni.c clab_engine.c ${PICOC_SRC})
target_compile_definitions(clab PRIVATE UNIX_HOST)
target_compile_options(clab PRIVATE -w -Wno-error=implicit-function-declaration -Wno-error=int-conversion -Wno-error=incompatible-function-pointer-types -Wno-error=incompatible-pointer-types -O2 -fno-strict-aliasing -fwrapv -fno-delete-null-pointer-checks)
target_link_libraries(clab m log)
target_link_options(clab PRIVATE -Wl,-z,max-page-size=16384)
